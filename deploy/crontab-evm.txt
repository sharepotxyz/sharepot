# SharePot on Robinhood Chain testnet — lines to add to the `sharepot` user's crontab next to deploy/crontab.txt.
# Both scripts are idempotent and find session times from the NYSE calendar themselves. They sign with the EVM
# proposer key only (~/secrets/evm/proposer.key: a hex private key; it may create markets, propose results and crank
# payouts). The contract's admin key is not on this machine.
#   EVM_CHAIN picks the deployment in server/evm/chains.json (pot address, RPC, tokens).

# keep one market per listed Robinhood stock token open for betting
*/30 * * * * cd $S/apps/sharepot && EVM_CHAIN=robinhood-testnet DATA_DIR=$S/data EVM_PROPOSER_KEY_FILE=$S/secrets/evm/proposer.key SEED_MARKETS=0 flock -n $S/locks/evm-open-markets.lock node server/evm/open-markets.mjs >> $S/logs/evm-open-markets.log 2>&1

# propose results after the close, finalize after the dispute window, pay out and sweep (indexes positions first)
*/10 * * * * cd $S/apps/sharepot && EVM_CHAIN=robinhood-testnet DATA_DIR=$S/data EVM_PROPOSER_KEY_FILE=$S/secrets/evm/proposer.key flock -n $S/locks/evm-resolve.lock node server/evm/resolve.mjs >> $S/logs/evm-resolve.log 2>&1
